<script>
(function () {
    var btn = document.getElementById('prog-toggle-all');
    if (!btn) return;
    var sessions = Array.prototype.slice.call(document.querySelectorAll('.program .prog-collapse'));

    function syncLabel() {
        var allOpen = sessions.every(function (d) { return d.open; });
        btn.textContent = allOpen ? 'Collapse all sessions' : 'Expand all sessions';
        btn.setAttribute('aria-expanded', allOpen ? 'true' : 'false');
    }

    btn.addEventListener('click', function () {
        var allOpen = sessions.every(function (d) { return d.open; });
        sessions.forEach(function (d) { d.open = !allOpen; });
        syncLabel();
    });

    sessions.forEach(function (d) {
        d.addEventListener('toggle', syncLabel);
    });
})();
</script>
